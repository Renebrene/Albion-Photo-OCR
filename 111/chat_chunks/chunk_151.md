    # ============================================================

            / "data"
            / "input"

    ).classes(
        "flex-1"
    )

    output_input = ui.input(
        label="Результат",
        value=str(
            PROJECT
            / "data"
            / "output"
            / "daily"
        ),

    ).classes(

    )

    )

    stop_button = None

    # HELPERS
    # ============================================================

            value
        ).expanduser()

        return path

        text: str,
        color_class: str = "text-grey-6",

            else:

            # Если скрипт не напечатал Output,
            # пытаемся определить последний запуск
            # из стандартной папки результатов.
