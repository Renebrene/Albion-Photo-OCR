cat > "$PYFILE" <<'PYTHON'
import os
import csv
import shutil
from pathlib import Path

import numpy as np
import torch
from PIL import Image
from transformers import AutoImageProcessor, AutoModel

ROOT = Path(os.environ["SF_ROOT"])
TEST_DIR = Path(os.environ["SF_TEST_DIR"])
FP_DIR = Path(os.environ["SF_FP_DIR"])
OUT_DIR = Path(os.environ["SF_OUT_DIR"])

THRESHOLD = float(os.environ["SF_THRESHOLD"])

FINGERPRINTS = os.environ["SF_FINGERPRINTS"].split(",")

IMAGE_EXTENSIONS = {
    ".jpg",
    ".jpeg",
    ".png",
    ".webp",
    ".bmp",
    ".tif",
    ".tiff",
}

# ============================================================
# HELPERS
# ============================================================

def normalize(v):

    v = np.asarray(
        v,
        dtype=np.float32
    )

    norm = np.linalg.norm(v)

    if norm == 0:
        return v

    return v / norm

def load_centroid(fp_id):

    path = FP_DIR / fp_id

    centroid = np.load(
        path / "centroid.npy"
    ).astype(np.float32)

    centroid = centroid.reshape(-1)

    return normalize(centroid)

def get_images():

    result = []

    for p in sorted(
        TEST_DIR.rglob("*")
    ):

        if (
            p.is_file()
            and p.suffix.lower()
            in IMAGE_EXTENSIONS
        ):
            result.append(p)

    return result

# ============================================================
# MODEL
# ============================================================

print()
print("[SceneEmbedder] Loading DINOv2...")

device = torch.device(
    "cuda"
    if torch.cuda.is_available()
    else "cpu"
)

print(
    f"[SceneEmbedder] device={device}"
)

if device.type == "cuda":

    print(
        "[SceneEmbedder] GPU:",
        torch.cuda.get_device_name(0)
    )

processor = AutoImageProcessor.from_pretrained(
    "facebook/dinov2-base"
)

model = AutoModel.from_pretrained(
    "facebook/dinov2-base"
)

model.to(device)
model.eval()

print(
    "[SceneEmbedder] ready"
)

# ============================================================
# FINGERPRINTS
# ============================================================

print()
print("[Fingerprint] Loading V33 centroids...")

fp_vectors = {}

for fp in FINGERPRINTS:

    vec = load_centroid(fp)

    fp_vectors[fp] = vec

    print(
        f"  {fp}: {vec.shape[0]} dimensions"
    )

fp_matrix = np.stack(
    [
        fp_vectors[fp]
        for fp in FINGERPRINTS
    ]
)

print(
    f"[Fingerprint] {len(FINGERPRINTS)} fingerprints loaded"
)

# ============================================================
# TEST
# ============================================================

images = get_images()

print()
print(
    f"[TEST] images found: {len(images)}"
)

if not images:

    print(
        "ERROR: no images found"
    )

    raise SystemExit(1)

# ============================================================
# CSV
# ============================================================

csv_path = (
    OUT_DIR
    / "results.csv"
)

csv_file = open(
    csv_path,
    "w",
    newline="",
    encoding="utf-8"
)

writer = csv.writer(
    csv_file
)

writer.writerow([
    "filename",

    "top1",
    "top1_score",

    "top2",
    "top2_score",

    "top3",
    "top3_score",

    "top4",
    "top4_score",

    "top5",
    "top5_score",

    "status",
])

# ============================================================
# COUNTERS
# ============================================================

counts = {
    "00": 0,
    "78": 0,
    "79": 0,
    "80": 0,
    "81": 0,
    "82": 0,
    "83": 0,
    "84": 0,
    "85": 0,
}

# ============================================================
# CLASSIFICATION
# ============================================================

print()
print("============================================================")
print(" RESULTS — TOP-1 ONLY")
print("============================================================")
print()

for n, image_path in enumerate(
    images,
    1
):

    relative = image_path.relative_to(
        TEST_DIR
    )

    try:

        # ----------------------------------------------------
        # IMAGE
        # ----------------------------------------------------

        image = Image.open(
            image_path
        ).convert("RGB")

        inputs = processor(
            images=image,
            return_tensors="pt"
        )

        inputs = {
            k: v.to(device)
            for k, v in inputs.items()
        }

        # ----------------------------------------------------
        # DINO
        # ----------------------------------------------------

        with torch.inference_mode():

            output = model(
                **inputs
            )

            embedding = (
                output
                .last_hidden_state[:, 0, :]
            )

            embedding = (
                torch.nn.functional.normalize(
                    embedding,
                    p=2,
                    dim=1
                )
            )

        emb = (
            embedding[0]
            .detach()
            .cpu()
            .numpy()
        )

        # ----------------------------------------------------
        # COSINE SIMILARITY
        # ----------------------------------------------------

        scores = (
            fp_matrix @ emb
        )

        order = np.argsort(
            scores
        )[::-1]

        # ----------------------------------------------------
        # TOP 5
        # ----------------------------------------------------

        top = []

        for idx in order[:5]:

            fp = FINGERPRINTS[idx]

            score = float(
                scores[idx]
            )

            top.append(
                (
                    fp.replace(
                        "_v33",
                        ""
                    ),
                    score
                )
            )

        top1, score1 = top[0]
        top2, score2 = top[1]
        top3, score3 = top[2]
        top4, score4 = top[3]
        top5, score5 = top[4]

        # ----------------------------------------------------
        # DECISION
        #
        # NO MARGIN!
        #
        # TOP-1 wins unless similarity is below threshold.
        # ----------------------------------------------------

        if score1 >= THRESHOLD:

            predicted = top1
            status = "MATCH"

        else:

            predicted = "00"
            status = "UNKNOWN"

        counts[predicted] += 1

        # ----------------------------------------------------
        # COPY
        # ----------------------------------------------------

        destination = (
            OUT_DIR
            / predicted
            / image_path.name
        )

        if destination.exists():

            destination = (
                OUT_DIR
                / predicted
                / (
                    image_path.stem
                    + f"__{n}"
                    + image_path.suffix
                )
            )

        shutil.copy2(
            image_path,
            destination
        )

        # ----------------------------------------------------
        # CSV
        # ----------------------------------------------------

        writer.writerow([
            str(relative),

            top1,
            f"{score1:.6f}",

            top2,
            f"{score2:.6f}",

            top3,
            f"{score3:.6f}",

            top4,
            f"{score4:.6f}",

            top5,
            f"{score5:.6f}",

            status,
        ])

        # ----------------------------------------------------
        # CONSOLE
        # ----------------------------------------------------

        print(
            f"{n:4d}/{len(images):4d} "
            f"{str(relative):42s} "
            f"-> {predicted:>2} "
            f"score={score1:.4f} "
            f"TOP2={top2}:{score2:.4f} "
            f"{status}"
        )

    except Exception as e:

        counts["00"] += 1

        print(
            f"{n:4d}/{len(images):4d} "
            f"{str(relative):42s} "
            f"-> 00 "
            f"ERROR: {e}"
        )

        writer.writerow([
            str(relative),
            "",
            "",
            "",
            "",
            "",
            "",
            "",
            "",
            "",
            "",
            f"ERROR: {e}",
        ])

csv_file.close()

# ============================================================
# SUMMARY
# ============================================================

print()
print("============================================================")
print(" SUMMARY")
print("============================================================")

print(
    f"Total:       {len(images)}"
)

print(
    f"Threshold:   {THRESHOLD}"
)

print()

for ID in [
    "00",
    "78",
    "79",
    "80",
    "81",
    "82",
    "83",
    "84",
    "85",
]:

    print(
        f"  {ID:>2}: {counts[ID]}"
    )

print()

recognized = (
    len(images)
    - counts["00"]
)

print(
    f"Recognized:  {recognized}"
)

print(
    f"Unknown 00:  {counts['00']}"
)

print()

print(
    "CSV:"
)

print(
    csv_path
)

print()

## Самое важное — перспектива

Допустим оригинал:

```text
┌─────────────────────────────┐
│                             │
│       БОЛЬШАЯ КАРТИНКА      │
│                             │
│       ТЕКСТ РЕКЛАМЫ         │
│                             │
└─────────────────────────────┘
```

А камера видит:

```text
       __________________
      /                 /
     /   БОЛЬШАЯ       /
    /    КАРТИНКА     /
   /_________________/
```

Для обычного сравнения картинок это уже достаточно сильное искажение.

Но мы можем использовать **perspective correction / homography**.

Если найдены четыре угла рекламного поля:

```text
A ---------------- B
|                  |
|      BANNER      |
|                  |
D ---------------- C
```

преобразуем его обратно:

```text
A' ---------------- B'
|                   |
|      BANNER       |
|                   |
D' ---------------- C'
```

И сравниваем уже с оригинальным макетом.
