#!/usr/bin/env python3

import shutil
from pathlib import Path
import json
import numpy as np
from tqdm import tqdm
from PIL import Image

from app.core.embedder import SceneEmbedder

PROJECT = Path(__file__).resolve().parent.parent

BANNER_DIR = PROJECT / "data/input/banners"
SEARCH_DIR = PROJECT / "data/input/test"
OUTPUT_DIR = PROJECT / "data/output/found_banners"

TOP_K = 10

def cosine_similarity(a, b):
    a = a / np.linalg.norm(a)
    b = b / np.linalg.norm(b)
    return float(np.dot(a, b))

def get_images(folder):
    exts = {".jpg", ".jpeg", ".png", ".JPG", ".JPEG"}
    return [
        p for p in folder.rglob("*")
        if p.suffix in exts
    ]

def main():

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

    print("=" * 70)
    print("BANNER SEARCH v01")
    print("=" * 70)

    print("Loading SceneEmbedder...")

    embedder = SceneEmbedder()

    print("Collecting search images...")

    search_images = get_images(SEARCH_DIR)

    if not search_images:
        print("ERROR: no images in", SEARCH_DIR)
        return

    print(f"Images to search: {len(search_images)}")

print("\nCreating image embeddings...")

    database = []

    for img_path in tqdm(search_images):

        try:
            emb = embedder.embed_image(str(img_path))

            database.append(
                {
                    "path": img_path,
                    "embedding": emb
                }
            )

        except Exception as e:
            print(
                "ERROR:",
                img_path.name,
                e
            )

print("\nSearching banners...")

    banners = get_images(BANNER_DIR)

    if not banners:
        print("ERROR: no banners in", BANNER_DIR)
        return

for banner in banners:

        print("\nQUERY:", banner.name)

        try:
            query_emb = embedder.embed_image(str(banner))

        except Exception as e:
            print("Embedding error:", e)
            continue

results = []

        for item in database:

            score = cosine_similarity(
                query_emb,
                item["embedding"]
            )

            results.append(
                (
                    score,
                    item["path"]
                )
            )

results.sort(
            key=lambda x: x[0],
            reverse=True
        )

banner_output = (
            OUTPUT_DIR /
            banner.stem
        )

        banner_output.mkdir(
            parents=True,
            exist_ok=True
        )

report = []

        for rank, (score, img) in enumerate(
            results[:TOP_K],
            start=1
        ):

            print(
                rank,
                img.name,
                f"{score:.4f}"
            )

            shutil.copy2(
                img,
                banner_output / img.name
            )

            report.append(
                {
                    "rank": rank,
                    "file": str(img),
                    "score": score
                }

    a = np.asarray(a)
    b = np.asarray(b)

    a_norm = np.linalg.norm(a)
    b_norm = np.linalg.norm(b)

    if a_norm == 0 or b_norm == 0:
        return 0.0

    return float(np.dot(a, b) / (a_norm * b_norm))

    a = np.asarray(a, dtype="float32")
    b = np.asarray(b, dtype="float32")

    a_norm = np.linalg.norm(a)
    b_norm = np.linalg.norm(b)

    if a_norm == 0 or b_norm == 0:
        return 0.0

    return float(
        np.dot(a, b) /
        (a_norm * b_norm)
    )
