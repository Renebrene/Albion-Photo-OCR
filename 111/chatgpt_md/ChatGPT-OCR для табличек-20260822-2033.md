```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Any

import cv2

@dataclass
class OCRResult:
    text: str
    confidence: float
    bbox: list[list[int]] | None = None

class OCRPipeline:
    """
    Отдельный OCR-модуль.

    На данном этапе:
    - принимает фотографию;
    - передает её в PaddleOCR;
    - возвращает найденный текст.

    Поиск конкретной белой таблички и работа с SQL
    пока намеренно не реализуются.
    """

    def __init__(self, lang: str = "ru"):
        self.lang = lang
        self._ocr = None

    def _load(self):
        """Ленивая загрузка PaddleOCR."""
        if self._ocr is not None:
            return

        try:
            from paddleocr import PaddleOCR
        except ImportError as exc:
            raise RuntimeError(
                "PaddleOCR не установлен. "
                "Установите его командой: "
                "uv add paddleocr"
            ) from exc

        self._ocr = PaddleOCR(
            lang=self.lang,
        )

    def recognize(self, image_path: str | Path) -> list[OCRResult]:
        """
        Распознать весь текст на фотографии.

        Пока не выполняется специальный поиск таблички.
        Это базовый тест самого OCR.
        """

        self._load()

        image_path = Path(image_path)

        if not image_path.exists():
            raise FileNotFoundError(
                f"Файл изображения не найден: {image_path}"
            )

        image = cv2.imread(str(image_path))

        if image is None:
            raise RuntimeError(
                f"Не удалось открыть изображение: {image_path}"
            )

        result = self._ocr.predict(str(image_path))

        return self._parse_result(result)

    @staticmethod
    def _parse_result(result: Any) -> list[OCRResult]:
        """
        Преобразовать результат PaddleOCR
        в простой внутренний формат.
        """

        results: list[OCRResult] = []

        for item in result:
            try:
                data = item.json
                if callable(data):
                    data = data()
            except Exception:
                data = None

            if not data:
                continue

            results.extend(
                OCRPipeline._parse_json_result(data)
            )

        return results

    @staticmethod
    def _parse_json_result(data: Any) -> list[OCRResult]:
        results: list[OCRResult] = []

        if isinstance(data, dict):
            # PaddleOCR 3.x обычно возвращает данные внутри "res".
            res = data.get("res", data)

            if isinstance(res, dict):
                texts = res.get("rec_texts", [])
                scores = res.get("rec_scores", [])
                boxes = res.get("rec_polys", [])

                for i, text in enumerate(texts):
                    try:
                        confidence = float(scores[i])
                    except (IndexError, TypeError, ValueError):
                        confidence = 0.0

                    bbox = None

                    try:
                        bbox = [
                            [int(point[0]), int(point[1])]
                            for point in boxes[i]
                        ]
                    except (IndexError, TypeError, ValueError):
                        pass

                    results.append(
                        OCRResult(
                            text=str(text),
                            confidence=confidence,
                            bbox=bbox,
                        )
                    )

        elif isinstance(data, list):
            for item in data:
                results.extend(
                    OCRPipeline._parse_json_result(item)
                )

        return results

def recognize_image(
    image_path: str | Path,
    lang: str = "ru",
) -> list[OCRResult]:
    """
    Удобная функция для вызова OCR без создания объекта вручную.
    """

    pipeline = OCRPipeline(lang=lang)

    return pipeline.recognize(image_path)

if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(
        description="Тестовый запуск OCR"
    )

    parser.add_argument(
        "image",
        help="Путь к фотографии",
    )

    parser.add_argument(
        "--lang",
        default="ru",
        help="Язык OCR, по умолчанию ru",
    )

    args = parser.parse_args()

    results = recognize_image(
        args.image,
        lang=args.lang,
    )

    print()
    print("=" * 70)
    print("OCR RESULT")
    print("=" * 70)

    if not results:
        print("Текст не найден.")
    else:
        for i, item in enumerate(results, 1):
            print(
                f"{i:02d}. "
                f"text={item.text!r} "
                f"confidence={item.confidence:.4f} "
                f"bbox={item.bbox}"
            )

    print("=" * 70)
```

from __future__ import annotations

import argparse
import time
from dataclasses import dataclass
from pathlib import Path

from paddleocr import PaddleOCR

@dataclass
class OCRResult:
    text: str
    confidence: float
    bbox: list

class OCRPipeline:
    """
    Базовый OCR-модуль.

    На этом этапе:
    - принимает фотографию;
    - запускает PaddleOCR;
    - возвращает весь найденный текст.

    Поиск белой таблички и SQL пока НЕ реализованы.
    """

    def __init__(self, lang: str = "ru"):
        self.lang = lang

        print("[OCR] Initializing PaddleOCR...")
        self.ocr = PaddleOCR(
            lang=self.lang,
            use_gpu=False,
        )
        print("[OCR] Ready. Device=CPU")

    def recognize(self, image_path: str | Path) -> list[OCRResult]:
        image_path = Path(image_path)

        if not image_path.exists():
            raise FileNotFoundError(
                f"Image not found: {image_path}"
            )

        start = time.perf_counter()

        result = self.ocr.ocr(
            str(image_path),
            cls=True,
        )

        elapsed = time.perf_counter() - start

        results = []

        if result and result[0]:
            for line in result[0]:
                bbox = line[0]
                text = line[1][0]
                confidence = float(line[1][1])

                results.append(
                    OCRResult(
                        text=text,
                        confidence=confidence,
                        bbox=bbox,
                    )
                )

        print(f"[OCR] Image: {image_path.name}")
        print(f"[OCR] Time: {elapsed:.3f} sec")
        print(f"[OCR] Text regions: {len(results)}")

        return results

def main():
    parser = argparse.ArgumentParser(
        description="Basic PaddleOCR test"
    )

    parser.add_argument(
        "image",
        help="Path to image",
    )

    parser.add_argument(
        "--lang",
        default="ru",
        help="OCR language",
    )

    args = parser.parse_args()

    ocr = OCRPipeline(
        lang=args.lang,
    )

    results = ocr.recognize(
        args.image,
    )

    print()
    print("=" * 70)
    print("OCR RESULT")
    print("=" * 70)

    if not results:
        print("No text detected.")
    else:
        for i, item in enumerate(results, 1):
            print(
                f"{i:02d}. "
                f"text={item.text!r} "
                f"confidence={item.confidence:.4f}"
            )

    print("=" * 70)

if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
from pathlib import Path

import cv2

def detect_white_rectangles(
    image_path: str | Path,
    output_path: str | Path,
    min_area: int = 2000,
):
    image_path = Path(image_path)
    output_path = Path(output_path)

    image = cv2.imread(str(image_path))

    if image is None:
        raise FileNotFoundError(image_path)

    original = image.copy()

    # Уменьшаем изображение только для поиска кандидатов.
    # Сам оригинал при этом не изменяется.
    scale = 1.0

    max_width = 1600

    if image.shape[1] > max_width:
        scale = max_width / image.shape[1]

        image = cv2.resize(
            image,
            None,
            fx=scale,
            fy=scale,
            interpolation=cv2.INTER_AREA,
        )

    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # Белые области.
    _, mask = cv2.threshold(
        gray,
        190,
        255,
        cv2.THRESH_BINARY,
    )

    # Немного объединяем разрывы внутри/по краям таблички.
    kernel = cv2.getStructuringElement(
        cv2.MORPH_RECT,
        (7, 7),
    )

    mask = cv2.morphologyEx(
        mask,
        cv2.MORPH_CLOSE,
        kernel,
    )

    contours, _ = cv2.findContours(
        mask,
        cv2.RETR_EXTERNAL,
        cv2.CHAIN_APPROX_SIMPLE,
    )

    candidates = []

    for contour in contours:
        area = cv2.contourArea(contour)

        if area < min_area:
            continue

        perimeter = cv2.arcLength(
            contour,
            True,
        )

        if perimeter == 0:
            continue

        approx = cv2.approxPolyDP(
            contour,
            0.03 * perimeter,
            True,
        )

        x, y, w, h = cv2.boundingRect(contour)

        if w <= 0 or h <= 0:
            continue

        aspect = w / h

        # Табличка должна быть заметно прямоугольной.
        if aspect < 1.3 or aspect > 8.0:
            continue

        # Проверяем заполненность прямоугольника.
        rect_area = w * h
        fill_ratio = area / rect_area

        if fill_ratio < 0.45:
            continue

        candidates.append(
            {
                "contour": contour,
                "bbox": (x, y, w, h),
                "area": area,
                "aspect": aspect,
                "fill_ratio": fill_ratio,
                "corners": len(approx),
            }
        )

    # Сортируем крупные кандидаты первыми.
    candidates.sort(
        key=lambda item: item["area"],
        reverse=True,
    )

    # Рисуем кандидатов на уменьшенной фотографии.
    debug = image.copy()

    for index, candidate in enumerate(candidates):
        x, y, w, h = candidate["bbox"]

        cv2.rectangle(
            debug,
            (x, y),
            (x + w, y + h),
            (0, 0, 255),
            3,
        )

        label = (
            f"{index + 1} "
            f"A={candidate['area']:.0f} "
            f"R={candidate['aspect']:.2f}"
        )

        cv2.putText(
            debug,
            label,
            (x, max(25, y - 8)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            (0, 0, 255),
            2,
            cv2.LINE_AA,
        )

    output_path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    cv2.imwrite(
        str(output_path),
        debug,
    )

    return candidates

def main():
    parser = argparse.ArgumentParser(
        description="Detect white rectangular plate candidates"
    )

    parser.add_argument(
        "image",
        help="Input image",
    )

    parser.add_argument(
        "--output",
        default="data/ocr_debug/plate_candidates.jpg",
        help="Debug output image",
    )

    args = parser.parse_args()

    candidates = detect_white_rectangles(
        args.image,
        args.output,
    )

    print()
    print("=" * 70)
    print("WHITE RECTANGLE DETECTOR")
    print("=" * 70)

    print(f"Candidates: {len(candidates)}")

    for i, candidate in enumerate(candidates, 1):
        print(
            f"{i:02d}. "
            f"bbox={candidate['bbox']} "
            f"area={candidate['area']:.0f} "
            f"aspect={candidate['aspect']:.2f} "
            f"fill={candidate['fill_ratio']:.2f}"
        )

    print("=" * 70)
    print()
    print(f"Debug image: {args.output}")

if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import multiprocessing as mp
import time
from dataclasses import dataclass
from pathlib import Path

import cv2
from paddleocr import PaddleOCR

# ============================================================
# Настройки эксперимента
# ============================================================

GRID_ROWS = 10
GRID_COLS = 10

# Перекрытие соседних тайлов.
# Нужно, чтобы текст/табличка на границе не разрезались.
OVERLAP = 0.15

# Количество OCR-процессов.
# 0 = автоматически.
WORKERS = 0

@dataclass
class Tile:
    index: int
    row: int
    col: int
    x1: int
    y1: int
    x2: int
    y2: int

@dataclass
class OCRText:
    tile: Tile
    text: str
    confidence: float

# ============================================================
# Глобальный OCR для каждого worker-процесса
# ============================================================

_worker_ocr = None

def init_worker():
    global _worker_ocr

    _worker_ocr = PaddleOCR(
        lang="ru",
        use_gpu=False,
    )

# ============================================================
# Создание сетки
# ============================================================

def create_tiles(
    width: int,
    height: int,
) -> list[Tile]:

    tiles = []

    tile_w = width / GRID_COLS
    tile_h = height / GRID_ROWS

    index = 0

    for row in range(GRID_ROWS):
        for col in range(GRID_COLS):

            base_x1 = int(col * tile_w)
            base_y1 = int(row * tile_h)

            base_x2 = int((col + 1) * tile_w)
            base_y2 = int((row + 1) * tile_h)

            # Добавляем overlap.
            extra_x = int(tile_w * OVERLAP)
            extra_y = int(tile_h * OVERLAP)

            x1 = max(0, base_x1 - extra_x)
            y1 = max(0, base_y1 - extra_y)

            x2 = min(width, base_x2 + extra_x)
            y2 = min(height, base_y2 + extra_y)

            tiles.append(
                Tile(
                    index=index,
                    row=row,
                    col=col,
                    x1=x1,
                    y1=y1,
                    x2=x2,
                    y2=y2,
                )
            )

            index += 1

    return tiles

# ============================================================
# OCR одного тайла
# ============================================================

def process_tile(
    args,
) -> list[OCRText]:

    image_path, tile = args

    global _worker_ocr

    result = _worker_ocr.ocr(
        str(image_path),
        cls=True,
        det=True,
    )

    results = []

    if not result or not result[0]:
        return results

    for line in result[0]:

        bbox = line[0]
        text = str(line[1][0])
        confidence = float(line[1][1])

        results.append(
            OCRText(
                tile=tile,
                text=text,
                confidence=confidence,
            )
        )

    return results

# ============================================================
# ВАЖНО:
# PaddleOCR получает изображение целиком.
#
# Поэтому для тайла сначала создаём временный JPEG.
# ============================================================

def prepare_tile_files(
    image,
    tiles,
    temp_dir,
):
    temp_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

    paths = []

    for tile in tiles:

        crop = image[
            tile.y1:tile.y2,
            tile.x1:tile.x2,
        ]

        path = temp_dir / f"tile_{tile.index:03d}.jpg"

        cv2.imwrite(
            str(path),
            crop,
            [
                cv2.IMWRITE_JPEG_QUALITY,
                92,
            ],
        )

        paths.append(
            (path, tile)
        )

    return paths

# ============================================================
# Основной запуск
# ============================================================

def run_grid_ocr(
    image_path: str | Path,
    workers: int = 0,
):

    image_path = Path(image_path)

    if not image_path.exists():
        raise FileNotFoundError(
            image_path
        )

    image = cv2.imread(
        str(image_path)
    )

    if image is None:
        raise RuntimeError(
            f"Cannot open image: {image_path}"
        )

    height, width = image.shape[:2]

    print()
    print("=" * 70)
    print("GRID OCR")
    print("=" * 70)

    print(
        f"Image: {image_path.name}"
    )

    print(
        f"Resolution: {width} x {height}"
    )

    print(
        f"Grid: {GRID_COLS} x {GRID_ROWS}"
    )

    print(
        f"Tiles: {GRID_COLS * GRID_ROWS}"
    )

    tiles = create_tiles(
        width,
        height,
    )

    temp_dir = (
        Path("data/ocr_grid")
        / image_path.stem
    )

    tile_files = prepare_tile_files(
        image,
        tiles,
        temp_dir,
    )

    if workers <= 0:
        cpu_count = mp.cpu_count()

        # Не забиваем систему полностью.
        workers = max(
            1,
            min(cpu_count, 8),
        )

    print(
        f"Workers: {workers}"
    )

    print(
        f"Tile directory: {temp_dir}"
    )

    print()
    print("Starting OCR...")
    print()

    start = time.perf_counter()

    tasks = [
        (
            path,
            tile,
        )
        for path, tile in tile_files
    ]

    all_results = []

    # spawn безопаснее для PaddleOCR.
    ctx = mp.get_context("spawn")

    with ctx.Pool(
        processes=workers,
        initializer=init_worker,
    ) as pool:

        for results in pool.imap_unordered(
            process_tile,
            tasks,
        ):

            all_results.extend(
                results
            )

            for item in results:

                print(
                    f"[TILE {item.tile.index:03d}] "
                    f"row={item.tile.row} "
                    f"col={item.tile.col} "
                    f"confidence={item.confidence:.3f} "
                    f"text={item.text!r}"
                )

    elapsed = (
        time.perf_counter()
        - start
    )

    # ========================================================
    # Сохраняем результаты
    # ========================================================

    result_file = (
        Path("data/ocr_grid")
        / f"{image_path.stem}_results.txt"
    )

    with result_file.open(
        "w",
        encoding="utf-8",
    ) as f:

        f.write(
            f"IMAGE: {image_path}\n"
        )

        f.write(
            f"RESOLUTION: {width}x{height}\n"
        )

        f.write(
            f"GRID: {GRID_COLS}x{GRID_ROWS}\n"
        )

        f.write(
            f"WORKERS: {workers}\n"
        )

        f.write(
            f"TIME: {elapsed:.3f}\n\n"
        )

        for item in sorted(
            all_results,
            key=lambda x: x.tile.index,
        ):

            f.write(
                f"TILE {item.tile.index:03d} "
                f"row={item.tile.row} "
                f"col={item.tile.col} "
                f"bbox=("
                f"{item.tile.x1},"
                f"{item.tile.y1},"
                f"{item.tile.x2},"
                f"{item.tile.y2}) "
                f"confidence="
                f"{item.confidence:.4f} "
                f"text={item.text!r}\n"
            )

    print()
    print("=" * 70)
    print("GRID OCR FINISHED")
    print("=" * 70)

    print(
        f"Time: {elapsed:.3f} sec"
    )

    print(
        f"Text regions: "
        f"{len(all_results)}"
    )

    print(
        f"Results: {result_file}"
    )

    print("=" * 70)

    return all_results

# ============================================================
# CLI
# ============================================================

def main():

    parser = argparse.ArgumentParser(
        description="OCR image using 10x10 grid"
    )

    parser.add_argument(
        "image",
        help="Input image",
    )

    parser.add_argument(
        "--workers",
        type=int,
        default=0,
        help="Number of OCR workers",
    )

    args = parser.parse_args()

    run_grid_ocr(
        args.image,
        workers=args.workers,
    )

if __name__ == "__main__":
    main()
