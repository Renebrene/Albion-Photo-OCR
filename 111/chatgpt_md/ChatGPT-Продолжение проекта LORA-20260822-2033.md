```bash
import os
import sys
import csv
import shutil
from pathlib import Path

import numpy as np
import torch
from PIL import Image
from transformers import AutoImageProcessor, AutoModel

# ============================================================
# Configuration from environment
# ============================================================

ROOT = Path(os.environ["SF_ROOT"])
TEST_DIR = Path(os.environ["SF_TEST_DIR"])
FP_DIR = Path(os.environ["SF_FP_DIR"])
OUT_DIR = Path(os.environ["SF_OUT_DIR"])

THRESHOLD = float(os.environ["SF_THRESHOLD"])
MARGIN = float(os.environ["SF_MARGIN"])

FINGERPRINTS = os.environ["SF_FINGERPRINTS"].split(",")

IMAGE_EXTENSIONS = {
    ".jpg",
    ".jpeg",
    ".png",
    ".webp",
    ".bmp",
    ".tif",
    ".tiff",
}

# ============================================================
# Helpers
# ============================================================

def normalize(x):
    x = np.asarray(x, dtype=np.float32)
    norm = np.linalg.norm(x, axis=-1, keepdims=True)
    norm[norm == 0] = 1.0
    return x / norm

def load_fingerprint(fp_id):
    """
    Load fingerprint embeddings and create one normalized
    centroid representation.

    We deliberately use ONLY embeddings.npy / centroid.npy.
    Patch embeddings are NOT loaded.
    """

    fp_path = FP_DIR / fp_id

    centroid_file = fp_path / "centroid.npy"
    embeddings_file = fp_path / "embeddings.npy"

    if centroid_file.exists():
        centroid = np.load(centroid_file)

        # Some files may contain an extra dimension.
        centroid = np.asarray(centroid, dtype=np.float32).reshape(-1)

        return normalize(centroid)

    if embeddings_file.exists():
        embeddings = np.load(embeddings_file)

        embeddings = np.asarray(
            embeddings,
            dtype=np.float32
        )

        if embeddings.ndim == 1:
            centroid = embeddings
        else:
            centroid = embeddings.mean(axis=0)

        return normalize(centroid)

    raise RuntimeError(
        f"Cannot load fingerprint {fp_id}: "
        f"no centroid.npy or embeddings.npy"
    )

def get_images(folder):
    result = []

    for p in sorted(folder.rglob("*")):
        if p.is_file() and p.suffix.lower() in IMAGE_EXTENSIONS:
            result.append(p)

    return result

# ============================================================
# Load model
# ============================================================

print()
print("[SceneEmbedder] Loading DINOv2...")

device = torch.device(
    "cuda" if torch.cuda.is_available() else "cpu"
)

print(f"[SceneEmbedder] device={device}")

if device.type == "cuda":
    print(
        "[SceneEmbedder] GPU:",
        torch.cuda.get_device_name(0)
    )

processor = AutoImageProcessor.from_pretrained(
    "facebook/dinov2-base"
)

model = AutoModel.from_pretrained(
    "facebook/dinov2-base"
)

model.to(device)
model.eval()

print("[SceneEmbedder] model ready")

# ============================================================
# Load fingerprints
# ============================================================

print()
print("[Fingerprint] Loading fingerprints...")

fingerprints = {}

for fp_id in FINGERPRINTS:
    fingerprints[fp_id] = load_fingerprint(fp_id)

    print(
        f"  {fp_id}: dimension={len(fingerprints[fp_id])}"
    )

fp_ids = list(fingerprints.keys())

fp_matrix = np.stack(
    [fingerprints[x] for x in fp_ids]
)

print(
    f"[Fingerprint] loaded {len(fp_ids)} fingerprints"
)

# ============================================================
# Find test images
# ============================================================

images = get_images(TEST_DIR)

print()
print(f"[TEST] directory: {TEST_DIR}")
print(f"[TEST] images:    {len(images)}")

if not images:
    print("No images found.")
    sys.exit(0)

# ============================================================
# CSV
# ============================================================

csv_path = OUT_DIR / "results.csv"

csv_file = open(
    csv_path,
    "w",
    newline="",
    encoding="utf-8"
)

writer = csv.writer(csv_file)

writer.writerow([
    "filename",
    "predicted",
    "best_score",
    "second_score",
    "margin",
    "status",
])

# ============================================================
# Classification
# ============================================================

recognized = 0
unknown = 0

print()
print("============================================================")
print(" RESULTS")
print("============================================================")

for index, image_path in enumerate(images, start=1):

    relative_name = image_path.relative_to(TEST_DIR)

    try:
        image = Image.open(image_path).convert("RGB")

        inputs = processor(
            images=image,
            return_tensors="pt"
        )

        inputs = {
            k: v.to(device)
            for k, v in inputs.items()
        }

        with torch.inference_mode():
            outputs = model(**inputs)

            # DINOv2 CLS token
            embedding = outputs.last_hidden_state[:, 0, :]

            embedding = torch.nn.functional.normalize(
                embedding,
                p=2,
                dim=1
            )

        emb = embedding[0].detach().cpu().numpy()

        # cosine similarity because both vectors normalized
        scores = fp_matrix @ emb

        order = np.argsort(scores)[::-1]

        best_idx = order[0]

        best_id = fp_ids[best_idx]
        best_score = float(scores[best_idx])

        if len(order) > 1:
            second_score = float(scores[order[1]])
        else:
            second_score = 0.0

        margin = best_score - second_score

        # ----------------------------------------------------
        # Decision
        # ----------------------------------------------------

        if (
            best_score >= THRESHOLD
            and margin >= MARGIN
        ):
            predicted = best_id
            status = "MATCH"
            recognized += 1
        else:
            predicted = "00"
            status = "UNKNOWN"
            unknown += 1

        # ----------------------------------------------------
        # Destination
        # ----------------------------------------------------

        destination_dir = OUT_DIR / predicted

        # Preserve relative directory structure if needed.
        # To avoid filename collisions, prefix nested paths.
        safe_name = str(relative_name).replace(
            os.sep,
            "__"
        )

        destination = destination_dir / safe_name

            str(relative_name),
            predicted,
            f"{best_score:.6f}",
            f"{second_score:.6f}",
            f"{margin:.6f}",
            status,
        ])

        print(
            f"{index:4d}/{len(images):4d}  "
            f"{relative_name}  ->  "
            f"{predicted:>2}  "
            f"score={best_score:.4f}  "
            f"second={second_score:.4f}  "
            f"margin={margin:.4f}  "
            f"{status}"
        )

    except Exception as e:

        print(
            f"{index:4d}/{len(images):4d}  "
            f"{relative_name}  ->  ERROR: {e}"
        )

        writer.writerow([
            str(relative_name),
            "00",
            "",
            "",
            "",
            f"ERROR: {e}",
        ])

        unknown += 1

# ============================================================
# Summary
# ============================================================

print()
print("============================================================")
print(" SUMMARY")
print("============================================================")

print(f"Recognized:  {recognized}")

print("CSV:")
print(f"  {csv_path}")

print()
print("============================================================")
PYTHON
```

```bash
    norm = np.linalg.norm(v)

    if norm == 0:
        return v

    return v / norm

def load_fp(fp_id):

    path = FP_DIR / fp_id

    # V33 centroid is preferred.
    centroid_file = path / "centroid.npy"

    if centroid_file.exists():

        centroid = np.load(
            centroid_file
        ).astype(np.float32)

        centroid = centroid.reshape(-1)

        return normalize(centroid)

    # Fallback.
    embeddings = np.load(
        path / "embeddings.npy"
    ).astype(np.float32)

    if embeddings.ndim == 1:
        centroid = embeddings
    else:
        centroid = embeddings.mean(axis=0)

    return normalize(centroid)
```

Например, у нас есть конструкция:

```text
Конструкция 83
├── banner_001.jpg
├── banner_002.jpg
├── banner_003.jpg
├── banner_004.jpg
└── banner_005.jpg
```

При этом баннеры могут быть **совершенно разными по цвету и содержанию**, но система знает:

> Эти изображения относятся к конструкции №83.

Затем приходит новая фотография:

```text
photo.jpg
```

На ней уже стоит новый баннер:

```text
[ рекламная конструкция №83 ]
[ НОВАЯ рекламная картинка  ]
```

Scene Fingerprint может сказать:

```text
83 = 0.94
```

а Banner Search отдельно:

```text
похож на banner_003
similarity = 0.81
```

И мы получаем:

```text
Конструкция: 83
Баннер:      группа 003
```

---

## 2. Но я бы не ограничивался обычным similarity

Для баннеров интереснее сделать **три независимых механизма**.

### IMAGE

Сравнение самой картинки:

```text
DINO / CLIP embedding
        ↓
FAISS
        ↓
похожие баннеры
```

Например:

```text
новый баннер
     ↓
0.91 → banner_003
0.87 → banner_017
0.84 → banner_005
```

Это будет работать даже если картинка немного:

- обрезана;
- снята под углом;
- имеет блики;
- имеет другой масштаб;
- частично закрыта;
- сфотографирована с расстояния.

---

### TEXT

Здесь вообще открывается очень интересная возможность.

Можно сделать:

```text
фотография
     ↓
OCR
     ↓
"КОМПАНИЯ РОМАШКА"
"СКИДКА 30%"
"ТЕЛЕФОН 8..."
```

И искать по текстовой базе.

Например, баннер был:

```text
"ЖК СОЛНЕЧНЫЙ"
```

а через месяц дизайн изменился:

```text
"ЖК СОЛНЕЧНЫЙ"
+ новая фотография
+ другой фон
```

Image similarity может заметно упасть.

Но OCR всё равно скажет:

```text
ЖК СОЛНЕЧНЫЙ
```

и система поймёт, что это тот же рекламодатель/кампания.

---

## 3. Можно искать и по отдельным словам

Например пользователь говорит:

> Найди все фотографии, где есть баннер со словом «СОЛНЕЧНЫЙ».

Система:

```text
OCR
 ↓
текст
 ↓
поиск
 ↓
photo_018.jpg
photo_037.jpg
photo_112.jpg
photo_246.jpg
```

Причём можно искать не только точное совпадение.

Например:

```text
"СОЛНЕЧНЫЙ"
```

может найти:

```text
ЖК СОЛНЕЧНЫЙ
СОЛНЕЧНЫЙ ГОРОД
СОЛНЕЧНЫЙ КВАРТАЛ
```

# 4. А видео — вообще без проблем

Вот здесь я бы сделал отдельный режим:

```text
video.mp4
   ↓
кадры
   ↓
каждые N кадров
   ↓
Banner Detector
   ↓
OCR + Image Search
   ↓
результаты
```

Например видео 2 минуты:

```text

не анализируем 3600 кадров
```

А берём, например:

```text
1 кадр / 1 сек
```

Получаем:

```text
120 кадров
```

И анализируем только их.

## 5. Ещё лучше — сначала найти рекламные конструкции

Можно сделать:

```text
VIDEO
 ↓
кадр
 ↓
обнаружение рекламной конструкции
 ↓
crop баннера
 ↓
IMAGE + OCR
```

То есть не прогонять весь кадр через OCR.

Например:

```text
┌─────────────────────────────┐
│                             │
│       улица                 │
│                             │
│     ┌───────────────┐       │
│     │   БАННЕР      │       │
│     │  СКИДКА 30%   │       │
│     └───────────────┘       │
│                             │
└─────────────────────────────┘
```

Система сначала вырезает:

```text
banner_crop.jpg
```

и уже его анализирует.

Это **сильно повысит качество OCR и поиска картинки**.

# 6. И можно определить, что баннер менялся

Это особенно полезно именно для твоей задачи.

Например камера сняла одну конструкцию 20 раз:

```text

10:05 → banner A
10:10 → banner A
10:15 → banner B
10:20 → banner B
10:25 → banner B
```

Система может выдать:

```text

  banner A

03.08
  banner B

07.08
  banner C
```

То есть фактически получится **история рекламных кампаний по каждой конструкции**.

У нас уже есть **известная конструкция**.


а не всю базу.

Получаем очень быструю систему:

```text

               │

               │

               │
               ▼
       Banner database 83
          /     |      \
         /      |       \
       img     OCR      logo
```

## Самое важное — перспектива

Допустим оригинал:

```text
┌─────────────────────────────┐
│                             │
│       БОЛЬШАЯ КАРТИНКА      │
│                             │
│       ТЕКСТ РЕКЛАМЫ         │
│                             │
└─────────────────────────────┘
```

А камера видит:

```text
       __________________
      /                 /
     /   БОЛЬШАЯ       /
    /    КАРТИНКА     /
   /_________________/
```

Для обычного сравнения картинок это уже достаточно сильное искажение.

Но мы можем использовать **perspective correction / homography**.

Если найдены четыре угла рекламного поля:

```text
A ---------------- B
|                  |
|      BANNER      |
|                  |
D ---------------- C
```

преобразуем его обратно:

```text
A' ---------------- B'
|                   |
|      BANNER       |
|                   |
D' ---------------- C'
```

И сравниваем уже с оригинальным макетом.
